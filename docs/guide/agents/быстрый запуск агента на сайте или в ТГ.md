**Код Cloudflare Worker (две ручки: /chat и /image):**

```js
  async fetch(request, env) {
    // Простые CORS-заголовки
    const cors = {
      "Access-Control-Allow-Origin": "*",
      "Access-Control-Allow-Headers": "Content-Type, Authorization",
      "Access-Control-Allow-Methods": "POST, OPTIONS"
    };
    if (request.method === "OPTIONS") return new Response("ok", { headers: cors });

    const url = new URL(request.url);

    // === /chat: ведём диалог и выдаём текст креатива ===
    if (url.pathname === "/chat" && request.method === "POST") {
      const { history = [] } = await request.json(); // [{role:'user'|'assistant'|'system', content:'...'}, ...]
      const systemPrompt = `
Ты профессиональный маркетолог и контент-менеджер с опытом 10+ лет.
Твоя задача — создать креатив: вовлекающий текст + промпт для изображения.
Перед написанием текста СНАЧАЛА задай вопросы:
1) В каком стиле писать текст (тональность, длина)?
2) Какой копирайтинг-фреймворк использовать — предложи варианты с кратким описанием: AIDA, PAS, 4P, BAB.
3) Чей стиль речи использовать (разреши загрузку/вставку примеров).
4) Определи целевую аудиторию, её боли/потребности, канал размещения и желаемое действие (CTA).

После получения ответов:
— Сформируй "Текст" (живой, без штампов, с лёгким юмором, затронь боли и предложи решение, приведи короткий житейский пример).
— Сформируй "Промпт_картинки" для генерации изображения (без лишнего текста, описательно).
Формат ответа строго такой:
### Текст
...готовый пост...
### Промпт_картинки
...краткий промпт...
      `.trim();

      const messages = [
        { role: "system", content: systemPrompt },
        ...history
      ];

      const resp = await fetch("https://api.openai.com/v1/chat/completions", {
        method: "POST",
        headers: {
          "Authorization": `Bearer ${env.OPENAI_API_KEY}`,
          "Content-Type": "application/json"
        },
        body: JSON.stringify({
          model: "gpt-4o-mini",
          messages,
          temperature: 0.6
        })
      });

      const data = await resp.json();
      const reply = data?.choices?.[0]?.message?.content ?? "Извини, не удалось получить ответ.";
      return new Response(JSON.stringify({ reply }), { headers: { "Content-Type": "application/json", ...cors } });
    }

    // === /image: генерируем картинку по промпту ===
    if (url.pathname === "/image" && request.method === "POST") {
      const { prompt } = await request.json();
      const imgResp = await fetch("https://api.openai.com/v1/images/generations", {
        method: "POST",
        headers: {
          "Authorization": `Bearer ${env.OPENAI_API_KEY}`,
          "Content-Type": "application/json"
        },
        body: JSON.stringify({
          model: "gpt-image-1",
          prompt,
          size: "1024x1024"
        })
      });
      const imgData = await imgResp.json();
      const b64 = imgData?.data?.[0]?.b64_json;
      return new Response(JSON.stringify({ image_base64: b64 }), { headers: { "Content-Type": "application/json", ...cors } });
    }

    return new Response("Not found", { status: 404, headers: cors });
  }
}
```

```js
import 'dotenv/config';
import { Telegraf } from 'telegraf';
import OpenAI from 'openai';

const bot = new Telegraf(process.env.BOT_TOKEN);
const openai = new OpenAI({ apiKey: process.env.OPENAI_API_KEY });

const systemPrompt = `... (тот же, что в Worker) ...`;

bot.start((ctx) => ctx.reply('Привет! Я помогу собрать вводные и создать креатив (текст + картинка). О какой теме поговорим?'));

bot.on('text', async (ctx) => {
  try {
    // Простейшая память на пользователя (можно заменить на БД)
    const userText = ctx.message.text;
    const messages = [
      { role: 'system', content: systemPrompt },
      { role: 'user', content: userText }
    ];
    const chat = await openai.chat.completions.create({
      model: 'gpt-4o-mini',
      messages,
      temperature: 0.6
    });
    const reply = chat.choices[0].message.content;
    await ctx.reply(reply);

    const after = reply.split('### Промпт_картинки')[1];
    if (after) {
      const imgPrompt = after.trim();
      const img = await openai.images.generate({
        model: 'gpt-image-1',
        prompt: imgPrompt,
        size: '1024x1024'
      });
      const b64 = img.data[0].b64_json;
      await ctx.replyWithPhoto({ source: Buffer.from(b64, 'base64') });
    }
  } catch (e) {
    console.error(e);
    await ctx.reply('Упс, не удалось сгенерировать ответ. Попробуйте переформулировать сообщение.');
  }
});

bot.launch();
console.log('Bot started');
```

# 1. `worker.js` — код для Cloudflare Workers

Этот код принимает вопрос от пользователя → подставляет твой промпт → шлёт в OpenAI → возвращает ответ.

```javascript
export default {
  async fetch(request, env) {
    if (request.method === "OPTIONS") {
      return new Response(null, {
        headers: {
          "Access-Control-Allow-Origin": "*",
          "Access-Control-Allow-Methods": "POST, OPTIONS",
          "Access-Control-Allow-Headers": "Content-Type",
        },
      });
    }

    if (request.method === "POST") {
      try {
        const body = await request.json();
        const userInput = body.message;

        // Твой промпт
        const systemPrompt = `
Ты профессиональный маркетолог и контент-менеджер с опытом более 10 лет.
Твоя задача - создать креатив: вовлекающий текст (пост) + картинка на тему: [спроси у меня].
Сначала задай мне вопросы: 
- в каком стиле писать текст;
- какой фреймворк использовать (предложи несколько вариантов на выбор в формате: аббревиатура + краткое описание);
- чей стиль речи использовать (с возможностью загрузки файла с примерами текстов).
Перед написанием текста определи основную целевую аудиторию, её основные потребности и боли.
Пиши легко, живо, с юмором, избегай шаблонов.
`;

        const response = await fetch("https://api.openai.com/v1/chat/completions", {
          method: "POST",
          headers: {
            "Authorization": `Bearer ${env.OPENAI_API_KEY}`,
            "Content-Type": "application/json",
          },
          body: JSON.stringify({
            model: "gpt-4o-mini",
            messages: [
              { role: "system", content: systemPrompt },
              { role: "user", content: userInput }
            ],
          }),
        });

        const data = await response.json();
        const aiMessage = data.choices?.[0]?.message?.content || "Извини, ответа нет.";

        return new Response(JSON.stringify({ reply: aiMessage }), {
          headers: {
            "Content-Type": "application/json",
            "Access-Control-Allow-Origin": "*",
          },
        });
      } catch (err) {
        return new Response(JSON.stringify({ error: err.message }), {
          status: 500,
          headers: { "Content-Type": "application/json" },
        });
      }
    }

    return new Response("Method not allowed", { status: 405 });
  }
};
```

Мы немного доработаем твой `worker.js`, чтобы он принимал не только запросы с сайта, но и webhook от Telegram.

### Новый `worker.js` (с поддержкой Telegram):

```javascript
export default {
  async fetch(request, env) {
    if (request.method === "OPTIONS") {
      return new Response(null, {
        headers: {
          "Access-Control-Allow-Origin": "*",
          "Access-Control-Allow-Methods": "POST, OPTIONS",
          "Access-Control-Allow-Headers": "Content-Type",
        },
      });
    }

    if (request.method === "POST") {
      try {
        const url = new URL(request.url);

        // --- Ветка для Telegram webhook ---
        if (url.pathname === "/telegram") {
          const update = await request.json();
          const chatId = update.message?.chat?.id;
          const userMessage = update.message?.text;

          if (!chatId || !userMessage) {
            return new Response("No message", { status: 200 });
          }

          // Промпт
          const systemPrompt = `
Ты профессиональный маркетолог и контент-менеджер с опытом более 10 лет.
Твоя задача - создать креатив: вовлекающий текст (пост) + картинка на тему: [спроси у меня].
Сначала задай мне вопросы: 
- в каком стиле писать текст;
- какой фреймворк использовать (предложи несколько вариантов на выбор в формате: аббревиатура + краткое описание);
- чей стиль речи использовать (с возможностью загрузки файла с примерами текстов).
Перед написанием текста определи основную целевую аудиторию, её основные потребности и боли.
Пиши легко, живо, с юмором, избегай шаблонов.
`;

          const response = await fetch("https://api.openai.com/v1/chat/completions", {
            method: "POST",
            headers: {
              "Authorization": `Bearer ${env.OPENAI_API_KEY}`,
              "Content-Type": "application/json",
            },
            body: JSON.stringify({
              model: "gpt-4o-mini",
              messages: [
                { role: "system", content: systemPrompt },
                { role: "user", content: userMessage }
              ],
            }),
          });

          const data = await response.json();
          const aiMessage = data.choices?.[0]?.message?.content || "Извини, ответа нет.";

          // Отправляем ответ обратно в Telegram
          await fetch(`https://api.telegram.org/bot${env.TELEGRAM_BOT_TOKEN}/sendMessage`, {
            method: "POST",
            headers: { "Content-Type": "application/json" },
            body: JSON.stringify({ chat_id: chatId, text: aiMessage }),
          });

          return new Response("OK", { status: 200 });
        }

        // --- Ветка для сайта (Tilda) ---
        const body = await request.json();
        const userInput = body.message;

        const systemPrompt = `
Ты профессиональный маркетолог и контент-менеджер с опытом более 10 лет.
Твоя задача - создать креатив: вовлекающий текст (пост) + картинка на тему: [спроси у меня].
Сначала задай мне вопросы: 
- в каком стиле писать текст;
- какой фреймворк использовать (предложи несколько вариантов на выбор в формате: аббревиатура + краткое описание);
- чей стиль речи использовать (с возможностью загрузки файла с примерами текстов).
Перед написанием текста определи основную целевую аудиторию, её основные потребности и боли.
Пиши легко, живо, с юмором, избегай шаблонов.
`;

        const response = await fetch("https://api.openai.com/v1/chat/completions", {
          method: "POST",
          headers: {
            "Authorization": `Bearer ${env.OPENAI_API_KEY}`,
            "Content-Type": "application/json",
          },
          body: JSON.stringify({
            model: "gpt-4o-mini",
            messages: [
              { role: "system", content: systemPrompt },
              { role: "user", content: userInput }
            ],
          }),
        });

        const data = await response.json();
        const aiMessage = data.choices?.[0]?.message?.content || "Извини, ответа нет.";

        return new Response(JSON.stringify({ reply: aiMessage }), {
          headers: {
            "Content-Type": "application/json",
            "Access-Control-Allow-Origin": "*",
          },
        });
      } catch (err) {
        return new Response(JSON.stringify({ error: err.message }), {
          status: 500,
          headers: { "Content-Type": "application/json" },
        });
      }
    }

    return new Response("Method not allowed", { status: 405 });
  }
};
```
